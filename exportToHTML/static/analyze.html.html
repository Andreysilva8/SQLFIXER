<script>
    function onFileSelect(input) {
        const file = input.files[0];
        if (!file) return;
        document.getElementById('file-name').style.display = 'block';
        document.getElementById('file-label').textContent = file.name;
    }

    async function enviarArquivo() {
        const fileInput = document.getElementById("arquivo");
        const file = fileInput.files[0];
        if (!file) {
            document.getElementById("resultado").innerText =
                "// ERRO: Nenhum arquivo selecionado.\n// Selecione um script .sql antes de iniciar.";
            return;
        }

        const status = document.getElementById("scan-status");
        status.style.display = 'inline-flex';
        status.textContent = 'ESCANEANDO...';
        status.className = 'tag tag-amber';

        document.getElementById("resultado").innerText =
            "// Iniciando varredura heurística...\n// Carregando assinaturas de ameaças...\n// Processando: " + file.name;

        const formData = new FormData();
        formData.append("file", file);

        try {
            const response = await fetch("http://127.0.0.1:5000/analyze", {
                method: "POST",
                body: formData
            });
            const data = await response.json();
            document.getElementById("resultado").innerText =
                data.result || "// Nenhuma anomalia crítica encontrada.\n// Score de risco: BAIXO";
            status.textContent = 'CONCLUÍDO';
            status.className = 'tag tag-green';
        } catch (error) {
            document.getElementById("resultado").innerText =
                "// ERRO: Servidor de análise offline.\n// Verifique se o backend Flask está rodando\n// em http://127.0.0.1:5000";
            status.textContent = 'ERRO';
            status.className = 'tag tag-red';
        }
    }
</script>
