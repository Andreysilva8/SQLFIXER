<script>
    /* ── LISTAR USUÁRIOS ── */
    async function listar() {
        try {
            const res = await fetch("http://localhost:8080/usuarios");
            const usuarios = await res.json();
            const tabela = document.getElementById("tabela");
            document.getElementById("total-users").innerText = usuarios.length;

            if (usuarios.length === 0) {
                tabela.innerHTML = `<tr><td colspan="6" style="text-align:center;color:var(--text-dim);font-family:'DM Mono',monospace;font-size:0.75rem;padding:32px;">Nenhum usuário cadastrado.</td></tr>`;
                return;
            }

            tabela.innerHTML = "";
            usuarios.forEach(u => {
                let dataFmt = u.dataNascimento
                    ? new Date(u.dataNascimento).toLocaleDateString('pt-BR')
                    : '-';
                tabela.innerHTML += `
                    <tr> 
                        <td class="td-id">#${u.id}</td> 
                        <td class="td-name">${u.nome}</td> 
                        <td>${u.email}</td> 
                        <td> 
                            <span class="badge-active"> 
                                <span class="pulse" style="width:5px;height:5px;"></span> 
                                Ativo 
                            </span> 
                        </td> 
                        <td style="font-family:'DM Mono',monospace;font-size:0.8rem;">${dataFmt}</td> 
                        <td> 
                            <button class="btn-danger" onclick="deletar(${u.id})">EXCLUIR</button> 
                        </td> 
                    </tr> 
                `;
            });
        } catch (e) {
            document.getElementById("tabela").innerHTML = `
                <tr><td colspan="6" style="text-align:center;color:var(--text-dim);font-family:'DM Mono',monospace;font-size:0.75rem;padding:32px;"> 
                    Servidor Java offline · http://localhost:8080 
                </td></tr>`;
        }
    }

    /* ── SALVAR USUÁRIO ── */
    async function salvar() {
        const data = {
            nome:           document.getElementById("nome").value,
            email:          document.getElementById("email").value,
            senha:          document.getElementById("senha").value,
            sexo:           document.getElementById("sexo").value,
            dataNascimento: document.getElementById("dataNascimento").value
        };

        const msg = document.getElementById("form-msg");

        try {
            const res = await fetch("http://localhost:8080/usuarios", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify(data)
            });

            if (res.ok) {
                msg.style.display = 'block';
                msg.style.background = 'var(--emerald-soft)';
                msg.style.border = '1px solid rgba(0,232,150,0.2)';
                msg.style.color = 'var(--emerald)';
                msg.innerText = '✓ Usuário sincronizado com sucesso ao Sentinel.';
                listar();
            } else {
                throw new Error('Server error');
            }
        } catch (e) {
            msg.style.display = 'block';
            msg.style.background = 'var(--red-soft)';
            msg.style.border = '1px solid rgba(240,72,96,0.2)';
            msg.style.color = 'var(--red)';
            msg.innerText = '✗ Erro ao conectar. Servidor offline ou dados inválidos.';
        }

        setTimeout(() => { msg.style.display = 'none'; }, 4000);
    }

    /* ── DELETAR ── */
    async function deletar(id) {
        if (confirm("Deseja revogar o acesso deste usuário do sistema?")) {
            await fetch("http://localhost:8080/usuarios/" + id, { method: "DELETE" });
            listar();
        }
    }

    listar();
</script>
